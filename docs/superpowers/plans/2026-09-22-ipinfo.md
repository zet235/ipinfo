# ipinfo Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** A Raycast/Tinycast extension with one `List` command that shows the public IP (from `https://ip.zet.tw/json`) and every local interface that has an IPv4 address, one copyable value per row.

**Architecture:** Two dependency-free modules under `src/lib/` (`public.ts` fetches JSON, `local.ts` shells out to `networksetup`/`ifconfig`/`ipconfig` and parses the text) are unit-tested with `node --test`. `src/ipinfo.tsx` is the only file that imports `@raycast/api`; it calls the two modules and lays out three `List.Section`s.

**Tech Stack:** TypeScript 5, React 19 via `@raycast/api` 2.4.x (`ray build`), Node 26 built-in test runner with native type stripping (no ts-node/tsx), macOS CLI tools `networksetup`, `ifconfig`, `ipconfig`.

**Spec:** `docs/superpowers/specs/2026-09-22-ipinfo-design.md`

## Global Constraints

- Extension `name` is `ipinfo`; command `name` is `ipinfo`, `title` is `Show IP Info`, `mode` is `view`.
- `src/lib/*.ts` never import `@raycast/api` or React.
- All subprocesses use `execFileSync` (argv array), never a shell string.
- Public fetch timeout: 5000 ms default.
- Build with `npx ray build -e dist -o build`. Never `-e dev`.
- No MAC, IPv6, DNS, caching, interval refresh, preferences.
- Commit messages end with the two attribution lines given by the session (`Co-Authored-By` + `Claude-Session`).

```ts
export interface PublicInfo {
  ip: string;
  asn: string;
  org: string;
  country: string;
  country_code: string;
  city: string;
}

export const PUBLIC_URL = "https://ip.zet.tw/json";

const FIELDS = ["ip", "asn", "org", "country", "country_code", "city"] as const;

export async function fetchPublicInfo(opts: { timeoutMs?: number; fetchImpl?: typeof fetch } = {}): Promise<PublicInfo> {
  const { timeoutMs = 5000, fetchImpl = fetch } = opts;
  const controller = new AbortController();
  const timer = setTimeout(() => controller.abort(new Error(`ip.zet.tw timed out after ${timeoutMs} ms`)), timeoutMs);
  try {
    const res = await fetchImpl(PUBLIC_URL, { signal: controller.signal, headers: { Accept: "application/json" } });
    if (!res.ok) throw new Error(`HTTP ${res.status} from ip.zet.tw`);
    const data = (await res.json()) as Record<string, unknown>;
    const info = {} as Record<(typeof FIELDS)[number], string>;
    for (const key of FIELDS) {
      const v = data[key];
      if (typeof v !== "string") throw new Error(`malformed response: missing ${key}`);
      info[key] = v;
    }
    return info;
  } finally {
    clearTimeout(timer);
  }
}
```

```ts
import { execFileSync } from "node:child_process";

export type InterfaceKind = "wifi" | "ethernet" | "other";

export interface LocalInterface {
  device: string;
  kind: InterfaceKind;
  label: string;
  ipv4: string;
  ssid?: string;
  router?: string;
}

export type Runner = (file: string, args: string[]) => string;

const defaultRunner: Runner = (file, args) => execFileSync(file, args, { encoding: "utf8" });

/** `networksetup -listallhardwareports` → device → "Hardware Port" name. */
export function parseHardwarePorts(text: string): Map<string, string> {
  const ports = new Map<string, string>();
  let port: string | undefined;
  for (const line of text.split("\n")) {
    const p = line.match(/^Hardware Port: (.+)$/);
    if (p) {
      port = p[1].trim();
      continue;
    }
    const d = line.match(/^Device: (\S+)$/);
    if (d && port) {
      ports.set(d[1], port);
      port = undefined;
    }
  }
  return ports;
}

/** `ifconfig` → device → first IPv4 address; interfaces without one (and lo0) are omitted. */
export function parseIfconfig(text: string): Map<string, string> {
  const addrs = new Map<string, string>();
  let device: string | undefined;
  for (const line of text.split("\n")) {
    const head = line.match(/^([a-z0-9]+): flags=/);
    if (head) {
      device = head[1];
      continue;
    }
    const inet = line.match(/^\s+inet (\d+\.\d+\.\d+\.\d+)/);
    if (inet && device && device !== "lo0" && !addrs.has(device)) {
      addrs.set(device, inet[1]);
    }
  }
  return addrs;
}

/** `ipconfig getsummary <dev>` → SSID and Router if present. */
export function parseWifiSummary(text: string): { ssid?: string; router?: string } {
  const out: { ssid?: string; router?: string } = {};
  const ssid = text.match(/^\s*SSID : (.+)$/m);
  if (ssid) out.ssid = ssid[1].trim();
  const router = text.match(/^\s*Router : (\S+)$/m);
  if (router) out.router = router[1];
  return out;
}

function kindOf(portName: string | undefined): InterfaceKind {
  if (!portName) return "other";
  if (/Wi-Fi/i.test(portName)) return "wifi";
  if (/Ethernet/i.test(portName)) return "ethernet";
  return "other";
}

const ORDER: Record<InterfaceKind, number> = { wifi: 0, ethernet: 1, other: 2 };

export function getLocalInterfaces(run: Runner = defaultRunner): LocalInterface[] {
  const ports = parseHardwarePorts(run("networksetup", ["-listallhardwareports"]));
  const addrs = parseIfconfig(run("ifconfig", []));

  const result: LocalInterface[] = [];
  for (const [device, ipv4] of addrs) {
    const port = ports.get(device);
    const kind = kindOf(port);
    const iface: LocalInterface = { device, kind, label: kind === "other" ? device : port!, ipv4 };
    if (kind === "wifi") {
      try {
        Object.assign(iface, parseWifiSummary(run("ipconfig", ["getsummary", device])));
      } catch {
        // ipconfig is best-effort; the IP row is still useful without SSID
      }
    }
    result.push(iface);
  }

  return result.sort((a, b) => ORDER[a.kind] - ORDER[b.kind] || a.device.localeCompare(b.device));
}
```

```tsx
import { useEffect, useMemo, useState } from "react";
import { fetchPublicInfo, type PublicInfo } from "./lib/public";
import { getLocalInterfaces, type LocalInterface } from "./lib/local";

type PublicState = { status: "loading" } | { status: "ok"; info: PublicInfo } | { status: "error"; message: string };

interface Row {
  key: string;
  title: string;
  subtitle?: string;
  accessory?: string;
}

function publicRows(state: PublicState): Row[] {
  if (state.status === "loading") return [];
  if (state.status === "error") return [{ key: "pub-err", title: "Unavailable", subtitle: state.message }];
  const { ip, asn, org, city, country, country_code } = state.info;
  return [
    { key: "pub-ip", title: ip, accessory: "ip.zet.tw" },
    { key: "pub-asn", title: asn, subtitle: org },
    { key: "pub-loc", title: `${city}, ${country} (${country_code})` },
  ];
}

function interfaceRow(i: LocalInterface): Row {
  const parts: string[] = [];
  if (i.ssid) parts.push(`SSID: ${i.ssid}`);
  if (i.router) parts.push(`Router ${i.router}`);
  return {
    key: `if-${i.device}`,
    title: i.ipv4,
    subtitle: parts.join(" · ") || undefined,
    accessory: i.kind === "other" ? i.device : undefined,
  };
}

function asText(state: PublicState, ifaces: LocalInterface[]): string {
  const lines: string[] = [];
  if (state.status === "ok") {
    const p = state.info;
    lines.push(`Public IP: ${p.ip}`, `ASN: ${p.asn}`, `Org: ${p.org}`, `Location: ${p.city}, ${p.country} (${p.country_code})`);
  }
  for (const i of ifaces) {
    const extra = [i.ssid && `SSID ${i.ssid}`, i.router && `Router ${i.router}`].filter(Boolean).join(", ");
    lines.push(`${i.label} (${i.device}): ${i.ipv4}${extra ? ` [${extra}]` : ""}`);
  }
  return lines.join("\n");
}

export default function Command() {
  const [pub, setPub] = useState<PublicState>({ status: "loading" });
  const ifaces = useMemo(() => {
    try {
      return getLocalInterfaces();
    } catch {
      return [];
    }
  }, []);

  useEffect(() => {
    let cancelled = false;
    fetchPublicInfo()
      .then((info) => !cancelled && setPub({ status: "ok", info }))
      .catch((e: unknown) => !cancelled && setPub({ status: "error", message: e instanceof Error ? e.message : String(e) }));
    return () => {
      cancelled = true;
    };
  }, []);

  const allText = asText(pub, ifaces);
  const actions = (value: string) => (
    <ActionPanel>
      <Action.CopyToClipboard title="Copy" content={value} />
      <Action.CopyToClipboard
        title="Copy All as Text"
        content={allText}
        shortcut={{ modifiers: ["cmd", "shift"], key: "c" }}
      />
      <Action.OpenInBrowser title="Open ip.zet.tw" url="https://ip.zet.tw" icon={Icon.Globe} />
    </ActionPanel>
  );

  const renderRow = (r: Row) => (
    <List.Item
      key={r.key}
      title={r.title}
      subtitle={r.subtitle}
      accessories={r.accessory ? [{ text: r.accessory }] : undefined}
      actions={actions(r.title)}
    />
  );

  const named = ifaces.filter((i) => i.kind !== "other");
  const others = ifaces.filter((i) => i.kind === "other");

  return (
    <List navigationTitle="IP Info" isLoading={pub.status === "loading"} searchBarPlaceholder="Filter…">
      <List.Section title="Public">{publicRows(pub).map(renderRow)}</List.Section>
      {named.map((i) => (
        <List.Section key={i.device} title={`${i.label} (${i.device})`}>
          {renderRow(interfaceRow(i))}
        </List.Section>
      ))}
      {others.length > 0 && <List.Section title="Other Interfaces">{others.map((i) => renderRow(interfaceRow(i)))}</List.Section>}
    </List>
  );
}
```
